   819→// ═════════════════════════════════════════════════════════════════════
   820→// TABLE 28 : PHYSICAL_QR_CODES — QR codes individuels physiques
   821→// ═══════════════════════════════════════════════════════════════════
   822→// Chaque QR physique a un code d'activation unique (ex: QR-A7K9M2P3).
   823→// Le QR physique pointe vers /activate/[activation_code].
   824→// Une fois activé, il est lié à un QrCode dynamique.
   825→
   826→model PhysicalQrCode {
   827→  id                  String    @id @default(cuid())
   828→  batchId             String    @map("batch_id")
   829→  activationCode      String    @unique @map("activation_code") // ex: "QR-A7K9M2P3"
   830→  status              String    @default("inactive")             // 'inactive'|'active'|'lost'|'cancelled'
   831→  activatedByUserId   String?   @map("activated_by_user_id")
   832→  activatedAt         DateTime? @map("activated_at")
   833→  dynamicQrCodeId     String?   @unique @map("dynamic_qr_code_id")
   834→  designConfigJson    String    @default("{}") @map("design_config_json") // hérité du batch
   835→
   836→  // ─── Relations ───
   837→  batch             QrBatch @relation(fields: [batchId], references: [id], onDelete: Cascade)
   838→  activatedBy        User?    @relation("Activator", fields: [activatedByUserId], references: [id], onDelete: SetNull)
   839→  dynamicQrCode      QrCode?  @relation("LinkedQrCode", fields: [dynamicQrCodeId], references: [id], onDelete: SetNull)
   840→  activationLogs     ActivationLog[]
   841→
   842→  @@index([batchId])
   843→  @@index([activationCode])
   844→  @@index([status])
   845→  @@index([activatedByUserId])
   846→  @@map("physical_qr_codes")
   847→}
