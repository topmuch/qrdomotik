   108→function AuthForm() {
   109→  const [isLogin, setIsLogin] = useState(true);
   110→  const [loading, setLoading] = useState(false);
   111→  const [email, setEmail] = useState('');
   112→  const [password, setPassword] = useState('');
   113→  const [fullName, setFullName] = useState('');
   114→
   115→  const handleSubmit = async (e: React.FormEvent) => {
   116→    e.preventDefault();
   117→    setLoading(true);
   118→    try {
   119→      if (!isLogin) {
   120→        const res = await fetch('/api/auth/register', {
   121→          method: 'POST', headers: { 'Content-Type': 'application/json' },
   122→          body: JSON.stringify({ email, password, fullName }),
   123→        });
   124→        const data = await res.json();
   125→        if (!data.success) { toast.error(data.error); return; }
   126→        toast.success('Compte créé ! Connexion...');
   127→      }
   128→      const result = await signIn('credentials', { email, password, redirect: false });
   129→      if (result?.error) toast.error('Email ou mot de passe incorrect');
   130→    } catch { toast.error('Erreur de connexion'); }
   131→    finally { setLoading(false); }
   132→  };
   133→
   134→  return (
   135→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-slate-50 via-white to-emerald-50/30">
   136→      <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} className="w-full max-w-md">
   137→        <div className="text-center mb-8">
   138→          <div className="relative w-20 h-20 mx-auto mb-4">
   139→            <div className="absolute inset-0 rounded-2xl bg-gradient-to-br from-emerald-400 to-teal-500 opacity-20 blur-xl" />
   140→            <Image src="/qr-domotik-logo.png" alt="QR Domotik" width={80} height={80} className="relative rounded-2xl drop-shadow-lg" />
   141→          </div>
   142→          <h1 className="text-3xl font-bold tracking-tight">QR <span className="text-emerald-600">Domotik</span></h1>
   143→          <p className="text-muted-foreground mt-1">Votre maison, intelligemment connectée</p>
   144→        </div>
   145→        <Card className="border-slate-200 shadow-lg shadow-slate-200/50">
   146→          <CardHeader className="pb-4">
   147→            <CardTitle className="flex items-center gap-2 text-lg">
   148→              {isLogin ? <LogIn className="w-5 h-5" /> : <UserPlus className="w-5 h-5" />}
   149→              {isLogin ? 'Connexion' : 'Créer un compte'}
   150→            </CardTitle>
   151→            <CardDescription>{isLogin ? 'Connectez-vous pour accéder à vos maisons' : 'Créez votre compte et votre première maison'}</CardDescription>
   152→          </CardHeader>
   153→          <CardContent>
   154→            <form onSubmit={handleSubmit} className="space-y-4">
   155→              <AnimatePresence mode="wait">
   156→                {!isLogin && (
   157→                  <motion.div key="name" initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }} transition={{ duration: 0.2 }}>
   158→                    <div className="space-y-2"><Label htmlFor="fullName">Nom complet</Label><Input id="fullName" placeholder="Jean Dupont" value={fullName} onChange={(e) => setFullName(e.target.value)} required={!isLogin} /></div>
   159→                  </motion.div>
   160→                )}
   161→              </AnimatePresence>
   162→              <div className="space-y-2"><Label htmlFor="email">Email</Label><Input id="email" type="email" placeholder="vous@exemple.com" value={email} onChange={(e) => setEmail(e.target.value)} required /></div>
   163→              <div className="space-y-2"><Label htmlFor="password">Mot de passe</Label><Input id="password" type="password" placeholder="••••••••" value={password} onChange={(e) => setPassword(e.target.value)} required minLength={6} /></div>
   164→              <Button type="submit" className="w-full" disabled={loading}>
   165→                {loading ? (<span className="flex items-center gap-2"><span className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />{isLogin ? 'Connexion...' : 'Création...'}</span>) : isLogin ? 'Se connecter' : "S'inscrire"}
   166→              </Button>
   167→            </form>
   168→            <div className="mt-4 text-center">
   169→              <button type="button" onClick={() => setIsLogin(!isLogin)} className="text-sm text-muted-foreground hover:text-foreground transition-colors">
   170→                {isLogin ? 'Pas encore de compte ?' : 'Déjà un compte ?'}{' '}<span className="font-medium text-emerald-600 hover:underline">{isLogin ? "S'inscrire" : 'Se connecter'}</span>
   171→              </button>
   172→            </div>
   173→          </CardContent>
   174→        </Card>
   175→        <p className="text-center text-xs text-muted-foreground mt-6">Plateforme complète QR Domotik</p>
   176→      </motion.div>
   177→    </div>
   178→  );
   179→}

   182→// ROOM ICON SELECTOR
   183→// ═══════════════════════════════════════════════════════════════════════════
   184→function RoomIconSelector({ value, onChange }: { value: string; onChange: (icon: string) => void }) {
   185→  const icons = ROOM_ICONS.filter(Boolean);
   186→  return (
   187→    <div className="grid grid-cols-7 gap-2">
   188→      {icons.map((name) => {
   189→        const IconComp = ROOM_ICON_MAP[name] || Square;
   190→        const isActive = value === name;
   191→        return (
   192→          <button key={name} type="button" onClick={() => onChange(name)} title={name}
   193→            className={`p-2 rounded-lg border transition-all flex items-center justify-center ${isActive ? 'border-emerald-500 bg-emerald-50 text-emerald-700 ring-2 ring-emerald-200' : 'border-slate-200 hover:border-slate-300 text-slate-500 hover:text-slate-700'}`}>
   194→            <IconComp className="w-5 h-5" />
   195→          </button>
   196→        );
   197→      })}
   198→    </div>
   199→  );
   200→}

Tous les QR codes et pièces seront supprimés.</AlertDialogDescription></AlertDialogHeader><AlertDialogFooter><AlertDialogCancel>Annuler</AlertDialogCancel><AlertDialogAction onClick={handleDeleteHome} className="bg-destructive text-destructive-foreground hover:bg-destructive/90">Supprimer</AlertDialogAction></AlertDialogFooter></AlertDialogContent>
   915→      </AlertDialog>
   916→      <AlertDialog open={!!deleteRoomId} onOpenChange={(open) => !open && setDeleteRoomId(null)}>
   917→        <AlertDialogContent><AlertDialogHeader><AlertDialogTitle>Supprimer cette pièce ?</AlertDialogTitle><AlertDialogDescription>Les QR codes seront dissociés.</AlertDialogDescription></AlertDialogHeader><AlertDialogFooter><AlertDialogCancel>Annuler</AlertDialogCancel><AlertDialogAction onClick={handleDeleteRoom} className="bg-destructive text-destructive-foreground hover:bg-destructive/90">Supprimer</AlertDialogAction></AlertDialogFooter></AlertDialogContent>
   918→      </AlertDialog>
   919→      <AlertDialog open={!!deleteQrId} onOpenChange={(open) => !open && setDeleteQrId(null)}>

   921→      </AlertDialog>
   922→
   923→      {/* ─── Footer ─── */}
   924→      <footer className="border-t border-slate-200 bg-white mt-auto">
   925→        <div className="container mx-auto px-4 py-4">

   927→            <div className="flex items-center gap-2">
