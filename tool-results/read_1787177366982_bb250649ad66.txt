   482→// ═══════════════════════════════════════════════════════════════════════════════
   483→// source='local' → créée par un commerçant
   484→// source='scraped' → récupérée par le script Playwright (Carrefour, Auchan...)
   485→// keywordsJson: JSON array string → ["pain", "baguette", "viennoiserie"]
   486→
   487→model Promo {
   488→  id                    String    @id @default(cuid())
   489→  merchantId            String?   @map("merchant_id")    // NULL si promo scrapée grande surface
   490→  source                String    @default("local")      // 'local' | 'scraped'
   491→  title                 String
   492→  description           String?
   493→  imageUrl              String?   @map("image_url")
   494→  originalPrice         Float?    @map("original_price")
   495→  promoPrice            Float     @map("promo_price")
   496→  validFrom             DateTime? @map("valid_from")
   497→  validUntil            DateTime  @map("valid_until")
   498→  keywordsJson          String    @default("[]") @map("keywords_json") // JSON array
   499→  category              String?                              // Pour filtrage par catégorie
   500→  isFlashSale           Boolean   @default(false) @map("is_flash_sale")
   501→  flashSaleTriggeredAt  DateTime? @map("flash_sale_triggered_at")
   502→  flashSaleCost         Float?    @map("flash_sale_cost")        // 0.50 € payé par le commerçant
   503→  viewsCount            Int       @default(0) @map("views_count")
   504→  redemptionsCount      Int       @default(0) @map("redemptions_count")
   505→  createdAt             DateTime  @default(now()) @map("created_at")
   506→  updatedAt             DateTime  @updatedAt @map("updated_at")
   507→
   508→  // ─── Relations ───
   509→  merchant   Merchant?          @relation(fields: [merchantId], references: [id], onDelete: SetNull)
   510→  redemptions PromoRedemption[]
   511→
   512→  @@index([merchantId])
   513→  @@index([source])
   514→  @@index([validUntil])
   515→  @@index([isFlashSale])
   516→  @@index([category])
   517→  @@index([keywordsJson])
   518→  @@map("promos")
   519→}
