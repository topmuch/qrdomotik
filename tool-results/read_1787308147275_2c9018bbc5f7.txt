   818→   818→// TABLE 28 : PHYSICAL_QR_CODES — QR codes individuels physiques
   819→   819→// ═══════════════════════════════════════════════════════════════════
   820→   820→// Chaque QR physique a un code d'activation unique (ex: QR-A7K9M2P3).
   821→   821→// Le QR physique pointe vers /activate/[activation_code].
   822→   822→// Une fois activé, il est lié à un QrCode dynamique.
   823→   823→
   824→   824→model PhysicalQrCode {
   825→   825→  id                  String    @id @default(cuid())
   826→   826→  batchId             String    @map("batch_id")
   827→   827→  activationCode      String    @unique @map("activation_code") // ex: "QR-A7K9M2P3"
   828→   828→  status              String    @default("inactive")             // 'inactive'|'active'|'lost'|'cancelled'
   829→   829→  activatedByUserId   String?   @map("activated_by_user_id")
   830→   830→  activatedAt         DateTime? @map("activated_at")
   831→   831→  dynamicQrCodeId     String?   @unique @map("dynamic_qr_code_id")
   832→   832→  designConfigJson    String    @default("{}") @map("design_config_json") // hérité du batch
   833→   833→
   834→   834→  // ─── Relations ───
   835→   835→  batch             QrBatch @relation(fields: [batchId], references: [id], onDelete: Cascade)
   836→   836→  activatedBy        User?    @relation("Activator", fields: [activatedByUserId], references: [id], onDelete: SetNull)
   837→   837→  dynamicQrCode      QrCode?  @relation("LinkedQrCode", fields: [dynamicQrCodeId], references: [id], onDelete: SetNull)
   838→   838→  activationLogs     ActivationLog[]
   839→   839→
   840→   840→  @@index([batchId])
   841→   841→  @@index([activationCode])
   842→   842→  @@index([status])
   843→   843→  @@index([activatedByUserId])
   844→   844→  @@map("physical_qr_codes")
   845→   845→}
