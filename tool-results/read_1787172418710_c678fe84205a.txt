    84→function AuthForm() {
    85→  const [isLogin, setIsLogin] = useState(true);
    86→  const [loading, setLoading] = useState(false);
    87→  const [email, setEmail] = useState('');
    88→  const [password, setPassword] = useState('');
    89→  const [fullName, setFullName] = useState('');
    90→
    91→  const handleSubmit = async (e: React.FormEvent) => {
    92→    e.preventDefault();
    93→    setLoading(true);
    94→    try {
    95→      if (!isLogin) {
    96→        const res = await fetch('/api/auth/register', {
    97→          method: 'POST', headers: { 'Content-Type': 'application/json' },
    98→          body: JSON.stringify({ email, password, fullName }),
    99→        });
   100→        const data = await res.json();
   101→        if (!data.success) { toast.error(data.error); return; }
   102→        toast.success('Compte créé ! Connexion...');
   103→      }
   104→      const result = await signIn('credentials', { email, password, redirect: false });
   105→      if (result?.error) toast.error('Email ou mot de passe incorrect');
   106→    } catch { toast.error('Erreur de connexion'); }
   107→    finally { setLoading(false); }
   108→  };
   109→
   110→  return (
   111→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-slate-50 via-white to-emerald-50/30">
   112→      <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} className="w-full max-w-md">
   113→        <div className="text-center mb-8">
   114→          <div className="relative w-20 h-20 mx-auto mb-4">
   115→            <div className="absolute inset-0 rounded-2xl bg-gradient-to-br from-emerald-400 to-teal-500 opacity-20 blur-xl" />
   116→            <Image src="/qr-domotik-logo.png" alt="QR Domotik" width={80} height={80} className="relative rounded-2xl drop-shadow-lg" />
   117→          </div>
   118→          <h1 className="text-3xl font-bold tracking-tight">QR <span className="text-emerald-600">Domotik</span></h1>
   119→          <p className="text-muted-foreground mt-1">Votre maison, intelligemment connectée</p>
   120→        </div>
   121→        <Card className="border-slate-200 shadow-lg shadow-slate-200/50">
   122→          <CardHeader className="pb-4">
   123→            <CardTitle className="flex items-center gap-2 text-lg">
   124→              {isLogin ? <LogIn className="w-5 h-5" /> : <UserPlus className="w-5 h-5" />}
   125→              {isLogin ? 'Connexion' : 'Créer un compte'}
   126→            </CardTitle>
   127→            <CardDescription>{isLogin ? 'Connectez-vous pour accéder à vos maisons' : 'Créez votre compte et votre première maison'}</CardDescription>
   128→          </CardHeader>
   129→          <CardContent>
   130→            <form onSubmit={handleSubmit} className="space-y-4">
   131→              <AnimatePresence mode="wait">
   132→                {!isLogin && (
   133→                  <motion.div key="name" initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }} transition={{ duration: 0.2 }}>
   134→                    <div className="space-y-2"><Label htmlFor="fullName">Nom complet</Label><Input id="fullName" placeholder="Jean Dupont" value={fullName} onChange={(e) => setFullName(e.target.value)} required={!isLogin} /></div>
   135→                  </motion.div>
   136→                )}
   137→              </AnimatePresence>
   138→              <div className="space-y-2"><Label htmlFor="email">Email</Label><Input id="email" type="email" placeholder="vous@exemple.com" value={email} onChange={(e) => setEmail(e.target.value)} required /></div>
   139→              <div className="space-y-2"><Label htmlFor="password">Mot de passe</Label><Input id="password" type="password" placeholder="••••••••" value={password} onChange={(e) => setPassword(e.target.value)} required minLength={6} /></div>
   140→              <Button type="submit" className="w-full" disabled={loading}>
   141→                {loading ? (<span className="flex items-center gap-2"><span className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />{isLogin ? 'Connexion...' : 'Création...'}</span>) : isLogin ? 'Se connecter' : "S'inscrire"}
   142→              </Button>
   143→            </form>
   144→            <div className="mt-4 text-center">
   145→              <button type="button" onClick={() => setIsLogin(!isLogin)} className="text-sm text-muted-foreground hover:text-foreground transition-colors">
   146→                {isLogin ? 'Pas encore de compte ?' : 'Déjà un compte ?'}{' '}<span className="font-medium text-emerald-600 hover:underline">{isLogin ? "S'inscrire" : 'Se connecter'}</span>
   147→              </button>
   148→            </div>
   149→          </CardContent>
   150→        </Card>
   151→        <p className="text-center text-xs text-muted-foreground mt-6">Plateforme complète QR Domotik</p>
   152→      </motion.div>
   153→    </div>
   154→  );
   155→}

   160→function RoomIconSelector({ value, onChange }: { value: string; onChange: (icon: string) => void }) {
   161→  const icons = ROOM_ICONS.filter(Boolean);
   162→  return (
   163→    <div className="grid grid-cols-7 gap-2">
   164→      {icons.map((name) => {
   165→        const IconComp = ROOM_ICON_MAP[name] || Square;
   166→        const isActive = value === name;
   167→        return (
   168→          <button key={name} type="button" onClick={() => onChange(name)} title={name}
   169→            className={`p-2 rounded-lg border transition-all flex items-center justify-center ${isActive ? 'border-emerald-500 bg-emerald-50 text-emerald-700 ring-2 ring-emerald-200' : 'border-slate-200 hover:border-slate-300 text-slate-500 hover:text-slate-700'}`}>
   170→            <IconComp className="w-5 h-5" />
   171→          </button>
   172→        );
   173→      })}
   174→    </div>
   175→  );
   176→}
