    51→function AuthForm() {
    52→  const [isLogin, setIsLogin] = useState(true);
    53→  const [loading, setLoading] = useState(false);
    54→  const [email, setEmail] = useState('');
    55→  const [password, setPassword] = useState('');
    56→  const [fullName, setFullName] = useState('');
    57→
    58→  const handleSubmit = async (e: React.FormEvent) => {
    59→    e.preventDefault();
    60→    setLoading(true);
    61→
    62→    try {
    63→      if (!isLogin) {
    64→        // Register
    65→        const res = await fetch('/api/auth/register', {
    66→          method: 'POST',
    67→          headers: { 'Content-Type': 'application/json' },
    68→          body: JSON.stringify({ email, password, fullName }),
    69→        });
    70→        const data = await res.json();
    71→        if (!data.success) {
    72→          toast.error(data.error);
    73→          return;
    74→        }
    75→        toast.success('Compte créé ! Connexion en cours...');
    76→      }
    77→      // Login
    78→      const result = await signIn('credentials', {
    79→        email, password, redirect: false,
    80→      });
    81→      if (result?.error) {
    82→        toast.error('Email ou mot de passe incorrect');
    83→      }
    84→    } catch {
    85→      toast.error('Erreur de connexion');
    86→    } finally {
    87→      setLoading(false);
    88→    }
    89→  };
    90→
    91→  return (
    92→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-slate-50 via-white to-emerald-50/30">
    93→      <motion.div
    94→        initial={{ opacity: 0, scale: 0.95 }}
    95→        animate={{ opacity: 1, scale: 1 }}
    96→        transition={{ duration: 0.4 }}
    97→        className="w-full max-w-md"
    98→      >
    99→        {/* Logo */}
   100→        <div className="text-center mb-8">
   101→          <div className="relative w-20 h-20 mx-auto mb-4">
   102→            <div className="absolute inset-0 rounded-2xl bg-gradient-to-br from-emerald-400 to-teal-500 opacity-20 blur-xl" />
   103→            <Image src="/qr-domotik-logo.png" alt="QR Domotik" width={80} height={80} className="relative rounded-2xl drop-shadow-lg" />
   104→          </div>
   105→          <h1 className="text-3xl font-bold tracking-tight">
   106→            QR <span className="text-emerald-600">Domotik</span>
   107→          </h1>
   108→          <p className="text-muted-foreground mt-1">Votre maison, intelligemment connectée</p>
   109→        </div>
   110→
   111→        {/* Form Card */}
   112→        <Card className="border-slate-200 shadow-lg shadow-slate-200/50">
   113→          <CardHeader className="pb-4">
   114→            <CardTitle className="flex items-center gap-2 text-lg">
   115→              {isLogin ? <LogIn className="w-5 h-5" /> : <UserPlus className="w-5 h-5" />}
   116→              {isLogin ? 'Connexion' : 'Créer un compte'}
   117→            </CardTitle>
   118→            <CardDescription>
   119→              {isLogin
   120→                ? 'Connectez-vous pour accéder à vos maisons'
   121→                : 'Créez votre compte et votre première maison'}
   122→            </CardDescription>
   123→          </CardHeader>
   124→          <CardContent>
   125→            <form onSubmit={handleSubmit} className="space-y-4">
   126→              <AnimatePresence mode="wait">
   127→                {!isLogin && (
   128→                  <motion.div
   129→                    key="name"
   130→                    initial={{ opacity: 0, height: 0 }}
   131→                    animate={{ opacity: 1, height: 'auto' }}
   132→                    exit={{ opacity: 0, height: 0 }}
   133→                    transition={{ duration: 0.2 }}
   134→                  >
   135→                    <div className="space-y-2">
   136→                      <Label htmlFor="fullName">Nom complet</Label>
   137→                      <Input
   138→                        id="fullName"
   139→                        placeholder="Jean Dupont"
   140→                        value={fullName}
   141→                        onChange={(e) => setFullName(e.target.value)}
   142→                        required={!isLogin}
   143→                      />
   144→                    </div>
   145→                  </motion.div>
   146→                )}
   147→              </AnimatePresence>
   148→
   149→              <div className="space-y-2">
   150→                <Label htmlFor="email">Email</Label>
   151→                <Input
   152→                  id="email"
   153→                  type="email"
   154→                  placeholder="vous@exemple.com"
   155→                  value={email}
   156→                  onChange={(e) => setEmail(e.target.value)}
   157→                  required
   158→                />
   159→              </div>
   160→
   161→              <div className="space-y-2">
   162→                <Label htmlFor="password">Mot de passe</Label>
   163→                <Input
   164→                  id="password"
   165→                  type="password"
   166→                  placeholder="••••••••"
   167→                  value={password}
   168→                  onChange={(e) => setPassword(e.target.value)}
   169→                  required
   170→                  minLength={6}
   171→                />
   172→              </div>
   173→
   174→              <Button type="submit" className="w-full" disabled={loading}>
   175→                {loading ? (
   176→                  <span className="flex items-center gap-2">
   177→                    <span className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />
   178→                    {isLogin ? 'Connexion...' : 'Création...'}
   179→                  </span>
   180→                ) : isLogin ? 'Se connecter' : "S'inscrire"}
   181→              </Button>
   182→            </form>
   183→
   184→            <div className="mt-4 text-center">
   185→              <button
   186→                type="button"
   187→                onClick={() => setIsLogin(!isLogin)}
   188→                className="text-sm text-muted-foreground hover:text-foreground transition-colors"
   189→              >
   190→                {isLogin ? 'Pas encore de compte ?' : 'Déjà un compte ?'}{' '}
   191→                <span className="font-medium text-emerald-600 hover:underline">
   192→                  {isLogin ? "S'inscrire" : 'Se connecter'}
   193→                </span>
   194→              </button>
   195→            </div>
   196→          </CardContent>
   197→        </Card>
   198→
   199→        <p className="text-center text-xs text-muted-foreground mt-6">
   200→          Étape 2 — Authentification & Dashboard
   201→        </p>
   202→      </motion.div>
   203→    </div>
   204→  );
   205→}

   207→// ─── Room Icon Selector ───────────────────────────────────────────────
   208→function RoomIconSelector({ value, onChange }: { value: string; onChange: (icon: string) => void }) {
   209→  const icons = ROOM_ICONS.filter(Boolean);
   210→  return (
   211→    <div className="grid grid-cols-7 gap-2">
   212→      {icons.map((name) => {
   213→        const IconComp = ICON_MAP[name] || Square;
   214→        const isActive = value === name;
   215→        return (
   216→          <button
   217→            key={name}
   218→            type="button"
   219→            onClick={() => onChange(name)}
   220→            className={`p-2 rounded-lg border transition-all flex items-center justify-center ${
   221→              isActive
   222→                ? 'border-emerald-500 bg-emerald-50 text-emerald-700 ring-2 ring-emerald-200'
   223→                : 'border-slate-200 hover:border-slate-300 text-slate-500 hover:text-slate-700'
   224→            }`}
   225→            title={name}
   226→          >
   227→            <IconComp className="w-5 h-5" />
   228→          </button>
   229→        );
   230→      })}
   231→    </div>
   232→  );
   233→}

   236→function Dashboard() {

   245→  const [deleteHomeId, setDeleteHomeId] = useState<string | null>(null);
   246→  const [deleteRoomId, setDeleteRoomId] = useState<string | null>(null);
   247→  const [editingRoom, setEditingRoom] = useState<{ id: string; name: string; icon: string } | null>(null);
   248→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   249→
   250→  useEffect(() => {
   251→    store.refreshHomes();
   252→  }, []);
   253→
   254→  // ─── Handlers ───
   255→  const handleCreateHome = async () => {
   256→    if (!homeName.trim()) return;
   257→    const res = await fetch('/api/homes', {
   258→      method: 'POST',
   259→      headers: { 'Content-Type': 'application/json' },
   260→      body: JSON.stringify({ name: homeName, address: homeAddress || undefined }),
   261→    });
   262→    const data = await res.json();
   263→    if (data.success) {
   264→      toast.success(`Maison « ${data.data.name} » créée`);
   265→      setHomeName('');
   266→      setHomeAddress('');
   267→      setHomeDialogOpen(false);
   268→      store.refreshHomes();
   269→    } else {
   270→      toast.error(data.error);
   271→    }
   272→  };
   273→
   274→  const handleDeleteHome = async () => {
   275→    if (!deleteHomeId) return;
   276→    const res = await fetch(`/api/homes/${deleteHomeId}`, { method: 'DELETE' });
   277→    const data = await res.json();
   278→    if (data.success) {
   279→      toast.success('Maison supprimée');
   280→      store.refreshHomes();
   281→    } else {
   282→      toast.error(data.error);
   283→    }
   284→    setDeleteHomeId(null);
   285→  };
   286→
   287→  const handleCreateRoom = async () => {
   288→    if (!roomName.trim() || !store.selectedHomeId) return;
   289→    const res = await fetch('/api/rooms', {
   290→      method: 'POST',
   291→      headers: { 'Content-Type': 'application/json' },
   292→      body: JSON.stringify({ homeId: store.selectedHomeId, name: roomName, icon: roomIcon }),
   293→    });

   296→      toast.success(`Pièce « ${data.data.name} » créée`);
   297→      setRoomName('');
   298→      setRoomIcon('Square');
   299→      setRoomDialogOpen(false);
   300→      store.refreshRooms();
   301→      store.refreshHomes();
   302→    } else {
   303→      toast.error(data.error);
   304→    }
   305→  };
   306→
   307→  const handleUpdateRoom = async () => {
   308→    if (!editingRoom || !editingRoom.name.trim()) return;
   309→    const res = await fetch(`/api/rooms/${editingRoom.id}`, {
   310→      method: 'PATCH',
   311→      headers: { 'Content-Type': 'application/json' },
   312→      body: JSON.stringify({ name: editingRoom.name, icon: editingRoom.icon }),
   313→    });
   314→    const data = await res.json();
   315→    if (data.success) {
   316→      toast.success('Pièce mise à jour');
   317→      setEditingRoom(null);
   318→      store.refreshRooms();
   319→      store.refreshHomes();
   320→    } else {
   321→      toast.error(data.error);
   322→    }
   323→  };
   324→
   325→  const handleDeleteRoom = async () => {
   326→    if (!deleteRoomId) return;
   327→    const res = await fetch(`/api/rooms/${deleteRoomId}`, { method: 'DELETE' });
   328→    const data = await res.json();
   329→    if (data.success) {
   330→      toast.success('Pièce supprimée');
   331→      store.refreshRooms();
   332→      store.refreshHomes();
   333→    } else {
   334→      toast.error(data.error);
   335→    }
   336→    setDeleteRoomId(null);
   337→  };
   338→
   339→  const selectedHome = store.getSelectedHome();
   340→  const memberRole = selectedHome?.role;
   341→  const canManage = memberRole === 'owner' || memberRole === 'admin';
   342→
   343→  return (
   344→    <div className="min-h-screen flex flex-col bg-slate-50">
   345→      {/* ─── Top Nav ─── */}

   347→        <div className="flex items-center justify-between h-14 px-4">
   348→          <div className="flex items-center gap-3">
   349→            <Button
   350→              variant="ghost" size="icon" className="md:hidden"
   351→              onClick={() => setMobileMenuOpen(!mobileMenuOpen)}
   352→            >
   353→              {mobileMenuOpen ? <X className="w-5 h-5" /> : <Menu className="w-5 h-5" />}
   354→            </Button>
   355→            <div className="flex items-center gap-2">
   356→              <Image src="/qr-domotik-logo.png" alt="QR Domotik" width={28} height={28} className="rounded-md" />
   357→              <span className="font-bold text-lg hidden sm:inline">
   358→                QR <span className="text-emerald-600">Domotik</span>
   359→              </span>
   360→            </div>
   361→          </div>
   362→          <div className="flex items-center gap-2">
   363→            <Badge variant="secondary" className="hidden sm:flex items-center gap-1.5">
   364→              <LayoutDashboard className="w-3.5 h-3.5" />
   365→              Étape 2
   366→            </Badge>
   367→            <div className="flex items-center gap-2 text-sm">
   368→              <span className="hidden md:inline text-muted-foreground">{session?.user?.name}</span>

   370→                <LogOut className="w-4 h-4" />

   578→            </motion.div>
