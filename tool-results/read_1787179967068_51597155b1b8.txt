    93→function AuthForm() {
    94→  const [isLogin, setIsLogin] = useState(true);
    95→  const [loading, setLoading] = useState(false);
    96→  const [email, setEmail] = useState('');
    97→  const [password, setPassword] = useState('');
    98→  const [fullName, setFullName] = useState('');
    99→
   100→  const handleSubmit = async (e: React.FormEvent) => {
   101→    e.preventDefault();
   102→    setLoading(true);
   103→    try {
   104→      if (!isLogin) {
   105→        const res = await fetch('/api/auth/register', {
   106→          method: 'POST', headers: { 'Content-Type': 'application/json' },
   107→          body: JSON.stringify({ email, password, fullName }),
   108→        });
   109→        const data = await res.json();
   110→        if (!data.success) { toast.error(data.error); return; }
   111→        toast.success('Compte créé ! Connexion...');
   112→      }
   113→      const result = await signIn('credentials', { email, password, redirect: false });
   114→      if (result?.error) toast.error('Email ou mot de passe incorrect');
   115→    } catch { toast.error('Erreur de connexion'); }
   116→    finally { setLoading(false); }
   117→  };
   118→
   119→  return (
   120→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-slate-50 via-white to-emerald-50/30">
   121→      <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} className="w-full max-w-md">
   122→        <div className="text-center mb-8">
   123→          <div className="relative w-20 h-20 mx-auto mb-4">
   124→            <div className="absolute inset-0 rounded-2xl bg-gradient-to-br from-emerald-400 to-teal-500 opacity-20 blur-xl" />
   125→            <Image src="/qr-domotik-logo.png" alt="QR Domotik" width={80} height={80} className="relative rounded-2xl drop-shadow-lg" />
   126→          </div>
   127→          <h1 className="text-3xl font-bold tracking-tight">QR <span className="text-emerald-600">Domotik</span></h1>
   128→          <p className="text-muted-foreground mt-1">Votre maison, intelligemment connectée</p>
   129→        </div>
   130→        <Card className="border-slate-200 shadow-lg shadow-slate-200/50">
   131→          <CardHeader className="pb-4">
   132→            <CardTitle className="flex items-center gap-2 text-lg">
   133→              {isLogin ? <LogIn className="w-5 h-5" /> : <UserPlus className="w-5 h-5" />}
   134→              {isLogin ? 'Connexion' : 'Créer un compte'}
   135→            </CardTitle>
   136→            <CardDescription>{isLogin ? 'Connectez-vous pour accéder à vos maisons' : 'Créez votre compte et votre première maison'}</CardDescription>
   137→          </CardHeader>
   138→          <CardContent>
   139→            <form onSubmit={handleSubmit} className="space-y-4">
   140→              <AnimatePresence mode="wait">
   141→                {!isLogin && (
   142→                  <motion.div key="name" initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }} transition={{ duration: 0.2 }}>
   143→                    <div className="space-y-2"><Label htmlFor="fullName">Nom complet</Label><Input id="fullName" placeholder="Jean Dupont" value={fullName} onChange={(e) => setFullName(e.target.value)} required={!isLogin} /></div>
   144→                  </motion.div>
   145→                )}
   146→              </AnimatePresence>
   147→              <div className="space-y-2"><Label htmlFor="email">Email</Label><Input id="email" type="email" placeholder="vous@exemple.com" value={email} onChange={(e) => setEmail(e.target.value)} required /></div>
   148→              <div className="space-y-2"><Label htmlFor="password">Mot de passe</Label><Input id="password" type="password" placeholder="••••••••" value={password} onChange={(e) => setPassword(e.target.value)} required minLength={6} /></div>
   149→              <Button type="submit" className="w-full" disabled={loading}>
   150→                {loading ? (<span className="flex items-center gap-2"><span className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />{isLogin ? 'Connexion...' : 'Création...'}</span>) : isLogin ? 'Se connecter' : "S'inscrire"}
   151→              </Button>
   152→            </form>
   153→            <div className="mt-4 text-center">
   154→              <button type="button" onClick={() => setIsLogin(!isLogin)} className="text-sm text-muted-foreground hover:text-foreground transition-colors">
   155→                {isLogin ? 'Pas encore de compte ?' : 'Déjà un compte ?'}{' '}<span className="font-medium text-emerald-600 hover:underline">{isLogin ? "S'inscrire" : 'Se connecter'}</span>
   156→              </button>
   157→            </div>
   158→          </CardContent>
   159→        </Card>
   160→        <p className="text-center text-xs text-muted-foreground mt-6">Plateforme complète QR Domotik</p>
   161→      </motion.div>
   162→    </div>
   163→  );
   164→}

   169→function RoomIconSelector({ value, onChange }: { value: string; onChange: (icon: string) => void }) {
   170→  const icons = ROOM_ICONS.filter(Boolean);
   171→  return (
   172→    <div className="grid grid-cols-7 gap-2">
   173→      {icons.map((name) => {
   174→        const IconComp = ROOM_ICON_MAP[name] || Square;
   175→        const isActive = value === name;
   176→        return (
   177→          <button key={name} type="button" onClick={() => onChange(name)} title={name}
   178→            className={`p-2 rounded-lg border transition-all flex items-center justify-center ${isActive ? 'border-emerald-500 bg-emerald-50 text-emerald-700 ring-2 ring-emerald-200' : 'border-slate-200 hover:border-slate-300 text-slate-500 hover:text-slate-700'}`}>
   179→            <IconComp className="w-5 h-5" />
   180→          </button>
   181→        );
   182→      })}
   183→    </div>
   184→  );
   185→}
