   142→// SHOPPING LIST EDITOR
   143→// ═══════════════════════════════════════════════════════════════
   144→function ShoppingListEditor({ content, onChange }: { content: any; onChange: (c: any) => void }) {
   145→  const items: Array<{ id: string; text: string }> = Array.isArray(content.items) ? content.items : [];
   146→  const [newItem, setNewItem] = useState('');
   147→
   148→  const addItem = () => {
   149→    if (!newItem.trim()) return;
   150→    const id = crypto.randomUUID().slice(0, 8);
   151→    onChange({ ...content, items: [...items, { id, text: newItem.trim(), checked: false }] });
   152→    setNewItem('');
   153→  };
   154→
   155→  const removeItem = (id: string) => {
   156→    onChange({ ...content, items: items.filter((i) => i.id !== id) });
   157→  };
   158→
   159→  return (
   160→    <div className="space-y-4">
   161→      <div className="flex gap-2">
   162→        <Input
   163→          placeholder="Ajouter un article..."
   164→          value={newItem}
   165→          onChange={(e) => setNewItem(e.target.value)}
   166→          onKeyDown={(e) => e.key === 'Enter' && (e.preventDefault(), addItem())}
   167→          className="flex-1"
   168→        />
   169→        <Button size="sm" onClick={addItem} disabled={!newItem.trim()}><Plus className="w-4 h-4" /></Button>
   170→      </div>
   171→      {items.length === 0 ? (
   172→        <p className="text-sm text-muted-foreground text-center py-4">La liste est vide</p>
   173→      ) : (
   174→        <div className="space-y-2 max-h-60 overflow-y-auto">
   175→          {items.map((item) => (
   176→            <div key={item.id} className="flex items-center gap-2 p-2 rounded-lg bg-slate-50">
   177→              <span className="flex-1 text-sm">{item.text}</span>
   178→              <Button variant="ghost" size="icon" className="h-7 w-7 text-destructive" onClick={() => removeItem(item.id)}>
   179→                <Trash2 className="w-3.5 h-3.5" />
   180→              </Button>
   181→            </div>
   182→          ))}
   183→        </div>
   184→      )}
   185→      <p className="text-xs text-muted-foreground">Les visiteurs pourront cocher les articles en scannant le QR.</p>
   186→    </div>
   187→  );
   188→}

   197→  const addInstruction = () => {
   198→    if (!newLabel.trim()) return;
   199→    const value = newLabel.trim().toLowerCase().replace(/\s+/g, '_');
   200→    onChange({ ...content, predefinedInstructions: [...instructions, { label: newLabel.trim(), value }] });
   201→    setNewLabel('');
   202→  };
   203→
   204→  const removeInstruction = (idx: number) => {
   205→    onChange({ ...content, predefinedInstructions: instructions.filter((_, i) => i !== idx) });
   206→  };
   207→
   208→  return (
   209→    <div className="space-y-4">
   210→      <div className="space-y-2">
   211→        <Label>Message d&apos;accueil</Label>

   338→const MEAL_OPTIONS = [
   339→  { value: 'petit-dejeuner', label: 'Petit-déjeuner', icon: '☕' },

   345→function DailyMenuEditor({ content, onChange }: { content: any; onChange: (c: any) => void }) {
   346→  const meals: Array<{ id: string; meal: string; dish: string; notes?: string }> = Array.isArray(content.meals) ? content.meals : [];

   354→
   355→  const updateMeal = (idx: number, field: string, value: string) => {

   359→  };
   360→
   361→  const removeMeal = (idx: number) => {
   362→    onChange({ ...content, meals: meals.filter((_, i) => i !== idx) });
   363→  };
   364→
   365→  return (
   366→    <div className="space-y-4">
   367→      {meals.map((meal, idx) => {
   368→        const mealOpt = MEAL_OPTIONS.find((m) => m.value === meal.meal);

   377→              </Button>
   378→            </div>
   379→            <select
   380→              value={meal.meal}
   381→              onChange={(e) => updateMeal(idx, 'meal', e.target.value)}
   382→              className="w-full h-9 rounded-md border border-input bg-background px-3 py-1 text-sm"

   390→      })}
   391→      <Button variant="outline" size="sm" onClick={addMeal}><Plus className="w-4 h-4 mr-1.5" /> Ajouter un repas</Button>
   392→      <p className="text-xs text-muted-foreground">Le menu du jour sera affiché avec la date actuelle.</p>
   393→    </div>
   394→  );
   395→}

   663→// MAIN CONTENT EDITOR
   664→// ═══════════════════════════════════════════════════════════════
   665→export function ContentEditor({ qrId, type, initialContent, onSave }: ContentEditorProps) {
   666→  const [content, setContent] = useState<any>({});
   667→  const [saving, setSaving] = useState(false);
   668→  const [dirty, setDirty] = useState(false);
   669→
   670→  useEffect(() => {
   671→    try { setContent(JSON.parse(initialContent)); } catch { setContent({}); }
   672→  }, [initialContent]);
   673→
   674→  const handleChange = (newContent: any) => {
   675→    setContent(newContent);
   676→    setDirty(true);
   677→  };
   678→
   679→  const handleSave = async () => {
   680→    setSaving(true);
   681→    try {
   682→      await onSave(JSON.stringify(content));
   683→      setDirty(false);
   684→    } finally {
   685→      setSaving(false);
   686→    }
   687→  };
   688→
   689→  const renderEditor = () => {
   690→    switch (type) {
   691→      case 'wifi': return <WifiEditor content={content} onChange={handleChange} />;
   692→      case 'link': return <LinkEditor content={content} onChange={handleChange} />;
   693→      case 'info': return <InfoEditor content={content} onChange={handleChange} />;
   694→      case 'postit': return <PostitEditor content={content} onChange={handleChange} />;
   695→      case 'shopping_list': return <ShoppingListEditor content={content} onChange={handleChange} />;
   696→      case 'doorman': return <DoormanEditor content={content} onChange={handleChange} />;
   697→      case 'medication': return <MedicationEditor content={content} onChange={handleChange} />;
   698→      case 'chores': return <ChoresEditor content={content} onChange={handleChange} />;
   699→      case 'stock_dlc': return <StockDlcEditor content={content} onChange={handleChange} />;
   700→      case 'daily_menu': return <DailyMenuEditor content={content} onChange={handleChange} />;
   701→      case 'todo_list': return <TodoListEditor content={content} onChange={handleChange} />;
   702→      case 'guestbook': return <GuestbookEditor content={content} onChange={handleChange} />;
   703→      case 'energy_counter': return <EnergyCounterEditor content={content} onChange={handleChange} />;
   704→      case 'keys_tracker': return <KeysTrackerEditor content={content} onChange={handleChange} />;
   705→      case 'deep_cleaning': return <DeepCleaningEditor content={content} onChange={handleChange} />;
   706→      default: return <p className="text-sm text-muted-foreground">Type non supporté</p>;
   707→    }
   708→  };
   709→
   710→  return (
   711→    <div className="space-y-4">
   712→      {renderEditor()}
   713→      {dirty && (
   714→        <Button onClick={handleSave} disabled={saving} className="w-full">
   715→          {saving ? 'Enregistrement...' : 'Enregistrer le contenu'}
   716→        </Button>
   717→      )}
   718→    </div>
   719→  );
   720→}

   723→// STOCK & DLC PANEL (for dashboard)
   724→// ═══════════════════════════════════════════════════════════════
   725→export function StockDlcPanel({ homeId }: { homeId: string }) {
   726→  const [products, setProducts] = useState<any[]>([]);
   727→  const [newName, setNewName] = useState('');
   728→  const [newCategory, setNewCategory] = useState('autre');
   729→  const [loading, setLoading] = useState(true);
   730→  const [expandedId, setExpandedId] = useState<string | null>(null);
   731→  const [newInstanceDate, setNewInstanceDate] = useState('');
   732→
   733→  const refresh = async () => {
   734→    setLoading(true);
   735→    try {
   736→      const res = await fetch(`/api/products?homeId=${homeId}`);
   737→      const json = await res.json();
   738→      if (json.success) setProducts(json.data);
   739→    } finally { setLoading(false); }
   740→  };
   741→
   742→  useEffect(() => { refresh(); }, [homeId]);
   743→
   744→  const addProduct = async () => {
   745→    if (!newName.trim()) return;
   746→    const res = await fetch('/api/products', {
   747→      method: 'POST',
   748→      headers: { 'Content-Type': 'application/json' },
   749→      body: JSON.stringify({ homeId, name: newName, category: newCategory }),
   750→    });
   751→    const json = await res.json();
   752→    if (json.success) { setNewName(''); refresh(); }
   753→  };
   754→
   755→  const deleteProduct = async (id: string) => {
   756→    await fetch(`/api/products/${id}`, { method: 'DELETE' });
   757→    if (expandedId === id) setExpandedId(null);
   758→    refresh();
   759→  };
   760→
   761→  const addInstance = async (productId: string) => {
   762→    if (!newInstanceDate) return;
   763→    await fetch(`/api/products/${productId}`, {
   764→      method: 'PATCH',
   765→      headers: { 'Content-Type': 'application/json' },
   766→      body: JSON.stringify({ action: 'add_instance', expiryDate: newInstanceDate }),
   767→    });
   768→    setNewInstanceDate('');
   769→    refresh();
   770→  };
   771→
   772→  const consumeInstance = async (productId: string, instanceId: string) => {
   773→    await fetch(`/api/products/${productId}`, {
   774→      method: 'PATCH',
   775→      headers: { 'Content-Type': 'application/json' },
   776→      body: JSON.stringify({ action: 'consume_instance', instanceId }),
   777→    });
   778→    refresh();
   779→  };
   780→
   781→  const getStatusColor = (expiryDate: string) => {
   782→    const days = Math.ceil((new Date(expiryDate).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   783→    if (days < 0) return 'bg-gray-100 text-gray-500 line-through';
   784→    if (days <= 1) return 'bg-red-100 text-red-700';
   785→    if (days <= 3) return 'bg-amber-100 text-amber-700';
   786→    return 'bg-emerald-100 text-emerald-700';
   787→  };
   788→
   789→  const getStatusLabel = (expiryDate: string) => {
   790→    const days = Math.ceil((new Date(expiryDate).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   791→    if (days < 0) return `Expiré (J${days})`;
   792→    if (days === 0) return 'Expire aujourd\'hui';
   793→    if (days === 1) return 'Demain';
   794→    if (days <= 3) return `J+${days}`;
   795→    return `J+${days}`;
   796→  };
   797→
   798→  return (
   799→    <div className="space-y-4">
   800→      {/* Add product */}
   801→      <div className="flex gap-2">
   802→        <Input placeholder="Nom du produit" value={newName} onChange={(e) => setNewName(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && addProduct()} className="flex-1" />
   803→        <select value={newCategory} onChange={(e) => setNewCategory(e.target.value)} className="h-10 rounded-md border border-input bg-background px-3 py-2 text-sm">
   804→          {Object.entries(PRODUCT_CATEGORY_LABELS).map(([k, v]) => <option key={k} value={k}>{v}</option>)}
   805→        </select>
   806→        <Button size="sm" onClick={addProduct} disabled={!newName.trim()}><Plus className="w-4 h-4" /></Button>
   807→      </div>
   808→
   809→      {/* Product list */}
   810→      {loading ? (
   811→        <div className="flex items-center justify-center py-8"><div className="w-6 h-6 border-2 border-lime-300 border-t-lime-600 rounded-full animate-spin" /></div>
   812→      ) : products.length === 0 ? (
   813→        <p className="text-sm text-muted-foreground text-center py-8">Aucun produit suivi</p>
   814→      ) : (
   815→        <div className="space-y-2 max-h-96 overflow-y-auto">
   816→          {products.map((product) => (
   817→            <div key={product.id} className="rounded-lg border border-slate-200 bg-white overflow-hidden">
   818→              <button
   819→                onClick={() => setExpandedId(expandedId === product.id ? null : product.id)}
   820→                className="w-full text-left p-3 flex items-center gap-3 hover:bg-slate-50 transition-colors"
   821→              >
   822→                <Package className="w-4 h-4 text-lime-600 shrink-0" />
   823→                <div className="flex-1 min-w-0">
   824→                  <p className="text-sm font-medium truncate">{product.name}</p>
   825→                  <p className="text-xs text-muted-foreground">{PRODUCT_CATEGORY_LABELS[product.category as keyof typeof PRODUCT_CATEGORY_LABELS] || product.category} · Stock: {product.currentStock}</p>
   826→                </div>
   827→                {product.productInstances[0] && (
   828→                  <Badge variant="secondary" className={`text-xs shrink-0 ${getStatusColor(product.productInstances[0].expiryDate)}`}>
   829→                    {getStatusLabel(product.productInstances[0].expiryDate)}
   830→                  </Badge>
   831→                )}
   832→                <Trash2 className="w-3.5 h-3.5 text-muted-foreground hover:text-destructive shrink-0" onClick={(e) => { e.stopPropagation(); deleteProduct(product.id); }} />
   833→              </button>
   834→
   835→              {expandedId === product.id && (
   836→                <div className="border-t border-slate-100 p-3 space-y-3 bg-slate-50/50">
   837→                  <div className="flex gap-2 items-end">
   838→                    <div className="flex-1 space-y-1">
   839→                      <Label className="text-xs">Ajouter une date de péremption</Label>
   840→                      <Input type="date" value={newInstanceDate} onChange={(e) => setNewInstanceDate(e.target.value)} className="h-9" />
