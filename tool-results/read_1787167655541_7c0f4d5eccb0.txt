    82→function AuthForm() {
    83→  const [isLogin, setIsLogin] = useState(true);
    84→  const [loading, setLoading] = useState(false);
    85→  const [email, setEmail] = useState('');
    86→  const [password, setPassword] = useState('');
    87→  const [fullName, setFullName] = useState('');
    88→
    89→  const handleSubmit = async (e: React.FormEvent) => {
    90→    e.preventDefault();
    91→    setLoading(true);
    92→    try {
    93→      if (!isLogin) {
    94→        const res = await fetch('/api/auth/register', {
    95→          method: 'POST', headers: { 'Content-Type': 'application/json' },
    96→          body: JSON.stringify({ email, password, fullName }),
    97→        });
    98→        const data = await res.json();
    99→        if (!data.success) { toast.error(data.error); return; }
   100→        toast.success('Compte créé ! Connexion...');
   101→      }
   102→      const result = await signIn('credentials', { email, password, redirect: false });
   103→      if (result?.error) toast.error('Email ou mot de passe incorrect');
   104→    } catch { toast.error('Erreur de connexion'); }
   105→    finally { setLoading(false); }
   106→  };
   107→
   108→  return (
   109→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-slate-50 via-white to-emerald-50/30">
   110→      <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} className="w-full max-w-md">
   111→        <div className="text-center mb-8">
   112→          <div className="relative w-20 h-20 mx-auto mb-4">
   113→            <div className="absolute inset-0 rounded-2xl bg-gradient-to-br from-emerald-400 to-teal-500 opacity-20 blur-xl" />
   114→            <Image src="/qr-domotik-logo.png" alt="QR Domotik" width={80} height={80} className="relative rounded-2xl drop-shadow-lg" />
   115→          </div>
   116→          <h1 className="text-3xl font-bold tracking-tight">QR <span className="text-emerald-600">Domotik</span></h1>
   117→          <p className="text-muted-foreground mt-1">Votre maison, intelligemment connectée</p>
   118→        </div>
   119→        <Card className="border-slate-200 shadow-lg shadow-slate-200/50">
   120→          <CardHeader className="pb-4">
   121→            <CardTitle className="flex items-center gap-2 text-lg">
   122→              {isLogin ? <LogIn className="w-5 h-5" /> : <UserPlus className="w-5 h-5" />}
   123→              {isLogin ? 'Connexion' : 'Créer un compte'}
   124→            </CardTitle>
   125→            <CardDescription>{isLogin ? 'Connectez-vous pour accéder à vos maisons' : 'Créez votre compte et votre première maison'}</CardDescription>
   126→          </CardHeader>
   127→          <CardContent>
   128→            <form onSubmit={handleSubmit} className="space-y-4">
   129→              <AnimatePresence mode="wait">
   130→                {!isLogin && (
   131→                  <motion.div key="name" initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }} transition={{ duration: 0.2 }}>
   132→                    <div className="space-y-2"><Label htmlFor="fullName">Nom complet</Label><Input id="fullName" placeholder="Jean Dupont" value={fullName} onChange={(e) => setFullName(e.target.value)} required={!isLogin} /></div>
   133→                  </motion.div>
   134→                )}
   135→              </AnimatePresence>
   136→              <div className="space-y-2"><Label htmlFor="email">Email</Label><Input id="email" type="email" placeholder="vous@exemple.com" value={email} onChange={(e) => setEmail(e.target.value)} required /></div>
   137→              <div className="space-y-2"><Label htmlFor="password">Mot de passe</Label><Input id="password" type="password" placeholder="••••••••" value={password} onChange={(e) => setPassword(e.target.value)} required minLength={6} /></div>
   138→              <Button type="submit" className="w-full" disabled={loading}>
   139→                {loading ? (<span className="flex items-center gap-2"><span className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />{isLogin ? 'Connexion...' : 'Création...'}</span>) : isLogin ? 'Se connecter' : "S'inscrire"}
   140→              </Button>
   141→            </form>
   142→            <div className="mt-4 text-center">
   143→              <button type="button" onClick={() => setIsLogin(!isLogin)} className="text-sm text-muted-foreground hover:text-foreground transition-colors">
   144→                {isLogin ? 'Pas encore de compte ?' : 'Déjà un compte ?'}{' '}<span className="font-medium text-emerald-600 hover:underline">{isLogin ? "S'inscrire" : 'Se connecter'}</span>
   145→              </button>
   146→            </div>
   147→          </CardContent>
   148→        </Card>
   149→        <p className="text-center text-xs text-muted-foreground mt-6">Étape 3 — Auth, Dashboard & QR Codes</p>
   150→      </motion.div>
   151→    </div>
   152→  );
   153→}

   158→function RoomIconSelector({ value, onChange }: { value: string; onChange: (icon: string) => void }) {
   159→  const icons = ROOM_ICONS.filter(Boolean);
   160→  return (
   161→    <div className="grid grid-cols-7 gap-2">
   162→      {icons.map((name) => {
   163→        const IconComp = ROOM_ICON_MAP[name] || Square;
   164→        const isActive = value === name;
   165→        return (
   166→          <button key={name} type="button" onClick={() => onChange(name)} title={name}
   167→            className={`p-2 rounded-lg border transition-all flex items-center justify-center ${isActive ? 'border-emerald-500 bg-emerald-50 text-emerald-700 ring-2 ring-emerald-200' : 'border-slate-200 hover:border-slate-300 text-slate-500 hover:text-slate-700'}`}>
   168→            <IconComp className="w-5 h-5" />
   169→          </button>
   170→        );
   171→      })}
   172→    </div>
   173→  );
   174→}

   208→function Dashboard() {
   209→  const { data: session } = useSession();
   210→  const store = useHomeStore();
   211→  const [homeName, setHomeName] = useState('');
   212→  const [homeAddress, setHomeAddress] = useState('');
   213→  const [roomName, setRoomName] = useState('');
   214→  const [roomIcon, setRoomIcon] = useState('Square');
   215→  const [homeDialogOpen, setHomeDialogOpen] = useState(false);
   216→  const [roomDialogOpen, setRoomDialogOpen] = useState(false);
   217→  const [deleteHomeId, setDeleteHomeId] = useState<string | null>(null);
   218→  const [deleteRoomId, setDeleteRoomId] = useState<string | null>(null);
   219→  const [deleteQrId, setDeleteQrId] = useState<string | null>(null);
   220→  const [editingRoom, setEditingRoom] = useState<{ id: string; name: string; icon: string } | null>(null);
   221→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   222→
   223→  // QR creation state
   224→  const [qrDialogOpen, setQrDialogOpen] = useState(false);
   225→  const [qrName, setQrName] = useState('');
   226→  const [qrType, setQrType] = useState<string>('wifi');
   227→  const [qrRoomId, setQrRoomId] = useState<string>('');
   228→  const [qrPin, setQrPin] = useState('');
   229→
   230→  // QR preview state
   231→  const [previewQr, setPreviewQr] = useState<QrCodeInfo | null>(null);
   232→  const [previewDialogOpen, setPreviewDialogOpen] = useState(false);
   233→  const [slugCopied, setSlugCopied] = useState(false);
   234→
   235→  useEffect(() => { store.refreshHomes(); }, []);
   236→
   237→  // ─── Home handlers ───
   238→  const handleCreateHome = async () => {
   239→    if (!homeName.trim()) return;
   240→    const res = await fetch('/api/homes', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name: homeName, address: homeAddress || undefined }) });
   241→    const data = await res.json();
   242→    if (data.success) { toast.success(`Maison « ${data.data.name} » créée`); setHomeName(''); setHomeAddress(''); setHomeDialogOpen(false); store.refreshHomes(); }
   243→    else toast.error(data.error);
   244→  };
   245→
   246→  const handleDeleteHome = async () => {
   247→    if (!deleteHomeId) return;
   248→    const res = await fetch(`/api/homes/${deleteHomeId}`, { method: 'DELETE' });
   249→    const data = await res.json();
   250→    if (data.success) { toast.success('Maison supprimée'); store.refreshHomes(); }
   251→    else toast.error(data.error);
   252→    setDeleteHomeId(null);
   253→  };
   254→
   255→  // ─── Room handlers ───
   256→  const handleCreateRoom = async () => {
   257→    if (!roomName.trim() || !store.selectedHomeId) return;
   258→    const res = await fetch('/api/rooms', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ homeId: store.selectedHomeId, name: roomName, icon: roomIcon }) });
   259→    const data = await res.json();
   260→    if (data.success) { toast.success(`Pièce « ${data.data.name} » créée`); setRoomName(''); setRoomIcon('Square'); setRoomDialogOpen(false); store.refreshRooms(); store.refreshHomes(); }
   261→    else toast.error(data.error);
   262→  };
   263→
   264→  const handleUpdateRoom = async () => {
   265→    if (!editingRoom || !editingRoom.name.trim()) return;
   266→    const res = await fetch(`/api/rooms/${editingRoom.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name: editingRoom.name, icon: editingRoom.icon }) });
   267→    const data = await res.json();
   268→    if (data.success) { toast.success('Pièce mise à jour'); setEditingRoom(null); store.refreshRooms(); store.refreshHomes(); }
   269→    else toast.error(data.error);
   270→  };
   271→
   272→  const handleDeleteRoom = async () => {
   273→    if (!deleteRoomId) return;
   274→    const res = await fetch(`/api/rooms/${deleteRoomId}`, { method: 'DELETE' });
   275→    const data = await res.json();
   276→    if (data.success) { toast.success('Pièce supprimée'); store.refreshRooms(); store.refreshHomes(); }
   277→    else toast.error(data.error);
   278→    setDeleteRoomId(null);
   279→  };
   280→
   281→  // ─── QR Code handlers ───
   282→  const handleCreateQr = async () => {
   283→    if (!qrName.trim() || !store.selectedHomeId) return;
   284→    const res = await fetch('/api/qr-codes', {
   285→      method: 'POST', headers: { 'Content-Type': 'application/json' },
   286→      body: JSON.stringify({
   287→        homeId: store.selectedHomeId,
   288→        name: qrName,
   289→        type: qrType,
   290→        roomId: qrRoomId || null,
   291→        pinCode: qrPin || undefined,
   292→        contentJson: DEFAULT_CONTENT[qrType] || '{}',
   293→      }),
   294→    });
   295→    const data = await res.json();
   296→    if (data.success) {
   297→      toast.success(`QR code « ${data.data.name} » créé`);
   298→      setQrName(''); setQrType('wifi'); setQrRoomId(''); setQrPin('');
   299→      setQrDialogOpen(false);
   300→      store.refreshQrCodes(); store.refreshHomes(); store.refreshRooms();
   301→      // Auto-open preview
   302→      setPreviewQr(data.data);
   303→      setPreviewDialogOpen(true);
   304→    } else toast.error(data.error);
   305→  };
   306→
   307→  const handleToggleQr = async (qr: QrCodeInfo) => {
   308→    const res = await fetch(`/api/qr-codes/${qr.id}`, {
   309→      method: 'PATCH', headers: { 'Content-Type': 'application/json' },
   310→      body: JSON.stringify({ isActive: !qr.isActive }),
   311→    });
   312→    const data = await res.json();
   313→    if (data.success) { toast.success(qr.isActive ? 'QR code désactivé' : 'QR code activé'); store.refreshQrCodes(); }
   314→    else toast.error(data.error);
   315→  };
   316→
   317→  const handleDeleteQr = async () => {
   318→    if (!deleteQrId) return;
   319→    const res = await fetch(`/api/qr-codes/${deleteQrId}`, { method: 'DELETE' });
   320→    const data = await res.json();
   321→    if (data.success) { toast.success('QR code supprimé'); store.refreshQrCodes(); store.refreshHomes(); }
   322→    else toast.error(data.error);
   323→    setDeleteQrId(null);
   324→  };
   325→
   326→  const handleCopySlug = useCallback((slug: string) => {
   327→    navigator.clipboard.writeText(`https://qrdomotik.com/r/${slug}`);
   328→    setSlugCopied(true);
   329→    setTimeout(() => setSlugCopied(false), 2000);
   330→  }, []);
   331→
   332→  const selectedHome = store.getSelectedHome();
   333→  const memberRole = selectedHome?.role;
   334→  const canManage = memberRole === 'owner' || memberRole === 'admin';
   335→
   336→  const filteredQrCodes = store.selectedRoomId
   337→    ? store.qrCodes.filter((q) => q.roomId === store.selectedRoomId)
   338→    : store.qrCodes;
   339→
   340→  return (
   341→    <div className="min-h-screen flex flex-col bg-slate-50">
   342→      {/* ─── Top Nav ─── */}
   343→      <header className="sticky top-0 z-40 bg-white border-b border-slate-200">
   344→        <div className="flex items-center justify-between h-14 px-4">
   345→          <div className="flex items-center gap-3">
   346→            <Button variant="ghost" size="icon" className="md:hidden" onClick={() => setMobileMenuOpen(!mobileMenuOpen)}>
   347→              {mobileMenuOpen ? <X className="w-5 h-5" /> : <Menu className="w-5 h-5" />}
   348→            </Button>
   349→            <div className="flex items-center gap-2">
   350→              <Image src="/qr-domotik-logo.png" alt="QR Domotik" width={28} height={28} className="rounded-md" />
   351→              <span className="font-bold text-lg hidden sm:inline">QR <span className="text-emerald-600">Domotik</span></span>
   352→            </div>
   353→          </div>
   354→          <div className="flex items-center gap-2">
   355→            <Badge variant="secondary" className="hidden sm:flex items-center gap-1.5">
   356→              <LayoutDashboard className="w-3.5 h-3.5" /> Étape 3
   357→            </Badge>
   358→            <span className="hidden md:inline text-sm text-muted-foreground">{session?.user?.name}</span>
   359→            <Button variant="ghost" size="icon" onClick={() => signOut()} title="Déconnexion"><LogOut className="w-4 h-4" /></Button>
   360→          </div>
   361→        </div>
   362→      </header>
   363→
   364→      <div className="flex-1 flex">
   365→        {/* ─── Sidebar ─── */}
   366→        <aside className={`fixed inset-y-0 left-0 top-14 z-30 w-72 bg-white border-r border-slate-200 transform transition-transform duration-200 md:static md:transform-none ${mobileMenuOpen ? 'translate-x-0' : '-translate-x-full md:translate-x-0'}`}>
   367→          <div className="flex flex-col h-full">
   368→            <div className="p-4 border-b border-slate-100 flex items-center justify-between">
   369→              <h2 className="font-semibold text-sm text-muted-foreground uppercase tracking-wider">Maisons</h2>
   370→              <Button size="icon" variant="ghost" className="h-7 w-7" onClick={() => { setHomeName(''); setHomeAddress(''); setHomeDialogOpen(true); }}><Plus className="w-4 h-4" /></Button>
   371→            </div>
   372→            <ScrollArea className="flex-1 max-h-[40vh]">
   373→              <div className="p-2 space-y-1">
   374→                {store.homes.map((home) => (
   375→                  <button key={home.id} onClick={() => { store.selectHome(home.id); setMobileMenuOpen(false); }}
   376→                    className={`w-full text-left px-3 py-2.5 rounded-lg flex items-center gap-3 transition-all ${store.selectedHomeId === home.id ? 'bg-emerald-50 text-emerald-700 border border-emerald-200' : 'hover:bg-slate-50 text-slate-700 border border-transparent'}`}>
   377→                    <Home className={`w-4 h-4 flex-shrink-0 ${store.selectedHomeId === home.id ? 'text-emerald-600' : 'text-slate-400'}`} />
   378→                    <div className="flex-1 min-w-0">
   379→                      <p className="text-sm font-medium truncate">{home.name}</p>
   380→                      <p className="text-xs text-muted-foreground">{home.roomsCount} pièce{home.roomsCount > 1 ? 's' : ''} · {home.qrCodesCount} QR</p>
   381→                    </div>
   382→                  </button>
   383→                ))}
   384→              </div>
   385→            </ScrollArea>
   386→
   387→            {/* Rooms section in sidebar */}
   388→            {selectedHome && (
   389→              <>
   390→                <div className="p-4 border-t border-slate-100 flex items-center justify-between">
   391→                  <h2 className="font-semibold text-sm text-muted-foreground uppercase tracking-wider">Pièces</h2>
   392→                  {canManage && (
