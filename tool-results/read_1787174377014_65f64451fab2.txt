    92→function AuthForm() {
    93→  const [isLogin, setIsLogin] = useState(true);
    94→  const [loading, setLoading] = useState(false);
    95→  const [email, setEmail] = useState('');
    96→  const [password, setPassword] = useState('');
    97→  const [fullName, setFullName] = useState('');
    98→
    99→  const handleSubmit = async (e: React.FormEvent) => {
   100→    e.preventDefault();
   101→    setLoading(true);
   102→    try {
   103→      if (!isLogin) {
   104→        const res = await fetch('/api/auth/register', {
   105→          method: 'POST', headers: { 'Content-Type': 'application/json' },
   106→          body: JSON.stringify({ email, password, fullName }),
   107→        });
   108→        const data = await res.json();
   109→        if (!data.success) { toast.error(data.error); return; }
   110→        toast.success('Compte créé ! Connexion...');
   111→      }
   112→      const result = await signIn('credentials', { email, password, redirect: false });
   113→      if (result?.error) toast.error('Email ou mot de passe incorrect');
   114→    } catch { toast.error('Erreur de connexion'); }
   115→    finally { setLoading(false); }
   116→  };
   117→
   118→  return (
   119→    <div className="min-h-screen flex items-center justify-center p-4 bg-gradient-to-br from-slate-50 via-white to-emerald-50/30">
   120→      <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} className="w-full max-w-md">
   121→        <div className="text-center mb-8">
   122→          <div className="relative w-20 h-20 mx-auto mb-4">
   123→            <div className="absolute inset-0 rounded-2xl bg-gradient-to-br from-emerald-400 to-teal-500 opacity-20 blur-xl" />
   124→            <Image src="/qr-domotik-logo.png" alt="QR Domotik" width={80} height={80} className="relative rounded-2xl drop-shadow-lg" />
   125→          </div>
   126→          <h1 className="text-3xl font-bold tracking-tight">QR <span className="text-emerald-600">Domotik</span></h1>
   127→          <p className="text-muted-foreground mt-1">Votre maison, intelligemment connectée</p>
   128→        </div>
   129→        <Card className="border-slate-200 shadow-lg shadow-slate-200/50">
   130→          <CardHeader className="pb-4">
   131→            <CardTitle className="flex items-center gap-2 text-lg">
   132→              {isLogin ? <LogIn className="w-5 h-5" /> : <UserPlus className="w-5 h-5" />}
   133→              {isLogin ? 'Connexion' : 'Créer un compte'}
   134→            </CardTitle>
   135→            <CardDescription>{isLogin ? 'Connectez-vous pour accéder à vos maisons' : 'Créez votre compte et votre première maison'}</CardDescription>
   136→          </CardHeader>
   137→          <CardContent>
   138→            <form onSubmit={handleSubmit} className="space-y-4">
   139→              <AnimatePresence mode="wait">
   140→                {!isLogin && (
   141→                  <motion.div key="name" initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }} transition={{ duration: 0.2 }}>
   142→                    <div className="space-y-2"><Label htmlFor="fullName">Nom complet</Label><Input id="fullName" placeholder="Jean Dupont" value={fullName} onChange={(e) => setFullName(e.target.value)} required={!isLogin} /></div>
   143→                  </motion.div>
   144→                )}
   145→              </AnimatePresence>
   146→              <div className="space-y-2"><Label htmlFor="email">Email</Label><Input id="email" type="email" placeholder="vous@exemple.com" value={email} onChange={(e) => setEmail(e.target.value)} required /></div>
   147→              <div className="space-y-2"><Label htmlFor="password">Mot de passe</Label><Input id="password" type="password" placeholder="••••••••" value={password} onChange={(e) => setPassword(e.target.value)} required minLength={6} /></div>
   148→              <Button type="submit" className="w-full" disabled={loading}>
   149→                {loading ? (<span className="flex items-center gap-2"><span className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />{isLogin ? 'Connexion...' : 'Création...'}</span>) : isLogin ? 'Se connecter' : "S'inscrire"}
   150→              </Button>
   151→            </form>
   152→            <div className="mt-4 text-center">
   153→              <button type="button" onClick={() => setIsLogin(!isLogin)} className="text-sm text-muted-foreground hover:text-foreground transition-colors">
   154→                {isLogin ? 'Pas encore de compte ?' : 'Déjà un compte ?'}{' '}<span className="font-medium text-emerald-600 hover:underline">{isLogin ? "S'inscrire" : 'Se connecter'}</span>
   155→              </button>
   156→            </div>
   157→          </CardContent>
   158→        </Card>
   159→        <p className="text-center text-xs text-muted-foreground mt-6">Plateforme complète QR Domotik</p>
   160→      </motion.div>
   161→    </div>
   162→  );
   163→}

   168→function RoomIconSelector({ value, onChange }: { value: string; onChange: (icon: string) => void }) {
   169→  const icons = ROOM_ICONS.filter(Boolean);
   170→  return (
   171→    <div className="grid grid-cols-7 gap-2">
   172→      {icons.map((name) => {
   173→        const IconComp = ROOM_ICON_MAP[name] || Square;
   174→        const isActive = value === name;
   175→        return (
   176→          <button key={name} type="button" onClick={() => onChange(name)} title={name}
   177→            className={`p-2 rounded-lg border transition-all flex items-center justify-center ${isActive ? 'border-emerald-500 bg-emerald-50 text-emerald-700 ring-2 ring-emerald-200' : 'border-slate-200 hover:border-slate-300 text-slate-500 hover:text-slate-700'}`}>
   178→            <IconComp className="w-5 h-5" />
   179→          </button>
   180→        );
   181→      })}
   182→    </div>
   183→  );
   184→}

   240→  // QR preview state

   256→  // ─── Home handlers ───
   257→  const handleCreateHome = async () => {
   258→    if (!homeName.trim()) return;
   259→    const res = await fetch('/api/homes', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name: homeName, address: homeAddress || undefined }) });
   260→    const data = await res.json();

   265→  const handleDeleteHome = async () => {
   266→    if (!deleteHomeId) return;
   267→    const res = await fetch(`/api/homes/${deleteHomeId}`, { method: 'DELETE' });
   268→    const data = await res.json();
   269→    if (data.success) { toast.success('Maison supprimée'); store.refreshHomes(); }

   271→    setDeleteHomeId(null);

   273→
   274→  // ─── Room handlers ───

   276→    if (!roomName.trim() || !store.selectedHomeId) return;
   277→    const res = await fetch('/api/rooms', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ homeId: store.selectedHomeId, name: roomName, icon: roomIcon }) });
   278→    const data = await res.json();
   279→    if (data.success) { toast.success(`Pièce « ${data.data.name} » créée`); setRoomName(''); setRoomIcon('Square'); setRoomDialogOpen(false); store.refreshRooms(); store.refreshHomes(); }
   280→    else toast.error(data.error);
   281→  };
   282→
   283→  const handleUpdateRoom = async () => {
   284→    if (!editingRoom || !editingRoom.name.trim()) return;
   285→    const res = await fetch(`/api/rooms/${editingRoom.id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name: editingRoom.name, icon: editingRoom.icon }) });

   287→    if (data.success) { toast.success('Pièce mise à jour'); setEditingRoom(null); store.refreshRooms(); store.refreshHomes(); }
   288→    else toast.error(data.error);
   289→  };
   290→
   291→  const handleDeleteRoom = async () => {
   292→    if (!deleteRoomId) return;
   293→    const res = await fetch(`/api/rooms/${deleteRoomId}`, { method: 'DELETE' });
   294→    const data = await res.json();
   295→    if (data.success) { toast.success('Pièce supprimée'); store.refreshRooms(); store.refreshHomes(); }

   360→      method: 'PATCH',

   366→      toast.success('Contenu mis à jour');

   369→      toast.error(data.error);

   373→  // ─── Doorman Toggle ───

   375→    const res = await fetch(`/api/qr-codes/${qr.id}`, {
   376→      method: 'PATCH',
   377→      headers: { 'Content-Type': 'application/json' },

   385→      toast.error(data.error);

   389→  const selectedHome = store.getSelectedHome();
   390→  const memberRole = selectedHome?.role;
   391→  const canManage = memberRole === 'owner' || memberRole === 'admin';
   392→
   393→  const filteredQrCodes = store.selectedRoomId
   394→    ? store.qrCodes.filter((q) => q.roomId === store.selectedRoomId)
   395→    : store.qrCodes;
   396→
   397→  return (
   398→    <div className="min-h-screen flex flex-col bg-slate-50">
   399→      {/* ─── Top Nav ─── */}

   407→              <Image src="/qr-domotik-logo.png" alt="QR Domotik" width={28} height={28} className="rounded-md" />

   409→            </div>
   410→          </div>
   411→          <div className="flex items-center gap-2">
